import { spawn } from 'child_process';
import fs from 'fs';
import path from 'path';

const SEEDS = [42, 137, 2718];
const SCENARIOS = [
  'BENIGN',
  'REPLAY_ATTACK',
  'CIRCUIT_BYPASS',
  'GHOST_CIRCUIT',
  'HANDSHAKE_SKIP',
  'PREMATURE_DATA',
  'CREATE_FLOOD'
];

const resultsDir = 'results/multi_seed_runs';
if (!fs.existsSync(resultsDir)) fs.mkdirSync(resultsDir, { recursive: true });

async function runScenario(seed, scenario) {
  return new Promise((resolve) => {
    const startTime = Date.now();
    const proc = spawn('node', [
      'experiments/shadow_harness_v2.mjs',
      '--seed', seed.toString(),
      '--scenario', scenario,
      '--timeout', '1800'
    ]);

    let stdout = '';
    let stderr = '';

    proc.stdout.on('data', (data) => {
      stdout += data.toString();
      process.stdout.write(`[Seed ${seed}] ${scenario}: ${data}`);
    });

    proc.stderr.on('data', (data) => {
      stderr += data.toString();
      process.stderr.write(`[ERROR] ${data}`);
    });

    proc.on('close', (code) => {
      const duration = ((Date.now() - startTime) / 1000).toFixed(2);
      const status = code === 0 ? 'SUCCESS' : 'FAILED';
      
      const result = {
        seed,
        scenario,
        status,
        duration: `${duration}s`,
        exitCode: code,
        timestamp: new Date().toISOString()
      };

      console.log(`✓ [${status}] Seed ${seed} - ${scenario} (${duration}s)`);
      resolve(result);
    });
  });
}

async function main() {
  const allResults = [];
  const statusLog = [];

  for (const seed of SEEDS) {
    console.log(`\n📊 Starting Seed ${seed}...`);
    for (const scenario of SCENARIOS) {
      const result = await runScenario(seed, scenario);
      allResults.push(result);
      statusLog.push(`${result.seed},${result.scenario},${result.status},${result.duration}`);
    }
  }

  // Sonuçları kaydet
  fs.writeFileSync(
    path.join(resultsDir, 'multi_seed_summary.json'),
    JSON.stringify(allResults, null, 2)
  );

  fs.writeFileSync(
    path.join(resultsDir, 'multi_seed_status.csv'),
    'SEED,SCENARIO,STATUS,DURATION\n' + statusLog.join('\n')
  );

  // Özet yazdır
  console.log('\n\n📈 MULTI-SEED SUMMARY:');
  const successCount = allResults.filter(r => r.status === 'SUCCESS').length;
  console.log(`✓ Success: ${successCount}/${allResults.length}`);
  console.log(`✗ Failed: ${allResults.length - successCount}/${allResults.length}`);
  
  // Seed 2718 özelinde kontrol
  const seed2718 = allResults.filter(r => r.seed === 2718);
  console.log(`\n🎯 Seed 2718 Results: ${seed2718.filter(r => r.status === 'SUCCESS').length}/${seed2718.length}`);
}

main().catch(console.error);

import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));

export class EventLevelMetrics {
  constructor() {
    this.predictions = [];
    this.groundTruth = [];
  }

  // Event-level confusion matrix
  calculateConfusionMatrix(predictions, groundTruth) {
    if (predictions.length !== groundTruth.length) {
      throw new Error('Predictions and ground truth length mismatch');
    }

    let tp = 0, fp = 0, tn = 0, fn = 0;

    for (let i = 0; i < predictions.length; i++) {
      const pred = predictions[i];
      const truth = groundTruth[i];

      if (pred === 1 && truth === 1) tp++;
      else if (pred === 1 && truth === 0) fp++;
      else if (pred === 0 && truth === 0) tn++;
      else if (pred === 0 && truth === 1) fn++;
    }

    return { tp, fp, tn, fn };
  }

  // Event-level metrics
  calculateMetrics(predictions, groundTruth) {
    const { tp, fp, tn, fn } = this.calculateConfusionMatrix(predictions, groundTruth);

    const precision = tp + fp === 0 ? 0 : tp / (tp + fp);
    const recall = tp + fn === 0 ? 0 : tp / (tp + fn);
    const f1 = precision + recall === 0 ? 0 : 2 * (precision * recall) / (precision + recall);
    const fpr = fp + tn === 0 ? 0 : fp / (fp + tn);
    const accuracy = (tp + tn) / (tp + fp + tn + fn);

    return {
      precision: parseFloat(precision.toFixed(4)),
      recall: parseFloat(recall.toFixed(4)),
      f1: parseFloat(f1.toFixed(4)),
      fpr: parseFloat(fpr.toFixed(4)),
      accuracy: parseFloat(accuracy.toFixed(4)),
      tp, fp, tn, fn
    };
  }

  // Event-level ROC-AUC
  calculateAUC(scores, groundTruth) {
    const n = scores.length;
    const pairs = scores.map((score, idx) => ({ score, truth: groundTruth[idx] }));
    pairs.sort((a, b) => b.score - a.score);

    let tp = 0, fp = 0;
    let tpPrev = 0, fpPrev = 0;
    let auc = 0;

    const positives = groundTruth.filter(x => x === 1).length;
    const negatives = groundTruth.filter(x => x === 0).length;

    for (const { truth } of pairs) {
      if (truth === 1) tp++;
      else fp++;

      auc += (tp - tpPrev) * (fpPrev + fp) / 2;
      tpPrev = tp;
      fpPrev = fp;
    }

    return auc / (positives * negatives);
  }

  // Per-event anomaly score
  calculateAnomalyScore(event, fsmState) {
    let score = 0;

    // Transition validity
    if (!this.isValidTransition(fsmState.current, event.type)) {
      score += 0.5;
    }

    // Event timing anomaly
    if (event.timestamp - fsmState.lastEventTime > 5000) {
      score += 0.2;
    }

    // Payload size anomaly
    if (event.payload && event.payload.length > 10000) {
      score += 0.15;
    }

    // Frequency anomaly
    if (fsmState.eventCount > 1000) {
      score += 0.15;
    }

    return Math.min(score, 1.0);
  }

  isValidTransition(currentState, eventType) {
    const validTransitions = {
      'CIRCUIT_CREATED': ['SEND_EXTEND', 'SEND_DATA'],
      'CIRCUIT_EXTENDED': ['SEND_EXTEND', 'SEND_DATA'],
      'TRANSMITTING': ['RECV_DATA', 'RECV_EXTENDED'],
      'CIRCUIT_CLOSED': ['CIRCUIT_CREATE']
    };

    return (validTransitions[currentState] || []).includes(eventType);
  }

  // Batch event-level analysis
  async analyzeEventsFromLog(logFile) {
    const events = [];
    const lines = fs.readFileSync(logFile, 'utf-8').split('\n');

    for (const line of lines) {
      if (!line.trim()) continue;
      try {
        const event = JSON.parse(line);
        events.push(event);
      } catch (e) {
        console.warn(`Skipping invalid JSON: ${line.substring(0, 50)}`);
      }
    }

    // Event-level predictions
    const predictions = events.map(e => e.isAnomaly ? 1 : 0);
    const groundTruth = events.map(e => e.groundTruth ? 1 : 0);

    const metrics = this.calculateMetrics(predictions, groundTruth);
    const auc = this.calculateAUC(
      events.map(e => e.anomalyScore || 0),
      groundTruth
    );

    return {
      eventCount: events.length,
      metrics,
      auc: parseFloat(auc.toFixed(4)),
      events: events.slice(0, 10) // Sample
    };
  }
}

// CLI kullanımı
if (import.meta.url === `file://${process.argv[1]}`) {
  const metricsEngine = new EventLevelMetrics();
  const logFile = process.argv[2] || 'results/events.jsonl';

}

export default EventLevelMetrics;

import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));

export class MetricsEngine {
  constructor(mode = 'event-level') {
    this.mode = mode; // 'event-level' or 'circuit-level'
    this.results = {};
  }

  // Event-level analysis
  analyzeEventLevel(events, scenario) {
    const predictions = [];
    const groundTruth = [];
    const anomalyScores = [];

    for (const event of events) {
      predictions.push(event.isAnomaly ? 1 : 0);
      groundTruth.push(event.groundTruth ? 1 : 0);
      anomalyScores.push(event.anomalyScore || 0);
    }

    return {
      mode: 'event-level',
      eventCount: events.length,
      metrics: this.calculateMetrics(predictions, groundTruth),
      auc: this.calculateAUC(anomalyScores, groundTruth),
      confusionMatrix: this.getConfusionMatrix(predictions, groundTruth)
    };
  }

  // Circuit-level analysis (legacy)
  analyzeCircuitLevel(circuits, scenario) {
    const predictions = [];
    const groundTruth = [];

    for (const circuit of circuits) {
      const hasAnomaly = circuit.events.some(e => e.isAnomaly);
      predictions.push(hasAnomaly ? 1 : 0);
      groundTruth.push(circuit.groundTruth ? 1 : 0);
    }

    return {
      mode: 'circuit-level',
      circuitCount: circuits.length,
      metrics: this.calculateMetrics(predictions, groundTruth),
      confusionMatrix: this.getConfusionMatrix(predictions, groundTruth)
    };
  }

  calculateMetrics(predictions, groundTruth) {
    const { tp, fp, tn, fn } = this.getConfusionMatrix(predictions, groundTruth);

    const precision = tp + fp === 0 ? 0 : tp / (tp + fp);
    const recall = tp + fn === 0 ? 0 : tp / (tp + fn);
    const f1 = precision + recall === 0 ? 0 : 2 * (precision * recall) / (precision + recall);
    const fpr = fp + tn === 0 ? 0 : fp / (fp + tn);
    const accuracy = (tp + tn) / (tp + fp + tn + fn);

    return {
      precision: parseFloat(precision.toFixed(4)),
      recall: parseFloat(recall.toFixed(4)),
      f1: parseFloat(f1.toFixed(4)),
      fpr: parseFloat(fpr.toFixed(4)),
      accuracy: parseFloat(accuracy.toFixed(4))
    };
  }

  getConfusionMatrix(predictions, groundTruth) {
    let tp = 0, fp = 0, tn = 0, fn = 0;

    for (let i = 0; i < predictions.length; i++) {
      if (predictions[i] === 1 && groundTruth[i] === 1) tp++;
      else if (predictions[i] === 1 && groundTruth[i] === 0) fp++;
      else if (predictions[i] === 0 && groundTruth[i] === 0) tn++;
      else if (predictions[i] === 0 && groundTruth[i] === 1) fn++;
    }

    return { tp, fp, tn, fn };
  }

  calculateAUC(scores, groundTruth) {
    const n = scores.length;
    const pairs = scores.map((score, idx) => ({ score, truth: groundTruth[idx] }));
    pairs.sort((a, b) => b.score - a.score);

    let tp = 0, fp = 0, auc = 0;
    const positives = groundTruth.filter(x => x === 1).length;
    const negatives = groundTruth.filter(x => x === 0).length;

    if (positives === 0 || negatives === 0) return 0;

    let tpPrev = 0, fpPrev = 0;
    for (const { truth } of pairs) {
      if (truth === 1) tp++;
      else fp++;
      auc += (tp - tpPrev) * (fpPrev + fp) / 2;
      tpPrev = tp;
      fpPrev = fp;
    }

    return auc / (positives * negatives);
  }

  async processLogFile(logFile) {
    const events = [];
    const lines = fs.readFileSync(logFile, 'utf-8').split('\n');

    for (const line of lines) {
      if (!line.trim()) continue;
      try {
        events.push(JSON.parse(line));
      } catch (e) {
        // Skip invalid lines
      }
    }

    if (this.mode === 'event-level') {
      return this.analyzeEventLevel(events, 'unknown');
    } else {
      // Group events by circuit
      const circuits = {};
      for (const event of events) {
        const cid = event.circuitId || 'unknown';
        if (!circuits[cid]) circuits[cid] = { id: cid, events: [] };
        circuits[cid].events.push(event);
      }
      return this.analyzeCircuitLevel(Object.values(circuits), 'unknown');
    }
  }
}

// CLI
if (import.meta.url === `file://${process.argv[1]}`) {
  const mode = process.argv[2] || 'event-level';
  const logFile = process.argv[3] || 'results/events.jsonl';

  const engine = new MetricsEngine(mode);
  engine.processLogFile(logFile).then(result => {
    console.log(`\n📊 ${mode.toUpperCase()} METRICS:`);
    console.log(JSON.stringify(result, null, 2));

    fs.writeFileSync(
      `results/metrics_${mode}.json`,
      JSON.stringify(result, null, 2)
    );
  });
}

export default MetricsEngine;
